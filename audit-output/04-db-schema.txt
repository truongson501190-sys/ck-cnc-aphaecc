-- ERP/WMS: Stock Ledger as single source of truth
-- Run in Supabase SQL Editor or via CLI

-- ---------------------------------------------------------------------------
-- stock_ledger (append-only)
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.stock_ledger (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  occurred_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  posted_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  warehouse_id TEXT NOT NULL,
  product_id TEXT NOT NULL,
  movement_type TEXT NOT NULL CHECK (
    movement_type IN (
      'IN', 'OUT', 'TRANSFER_OUT', 'TRANSFER_IN', 'ADJUSTMENT', 'OIL_OUT'
    )
  ),
  qty_delta NUMERIC(18, 4) NOT NULL CHECK (qty_delta <> 0),
  unit_cost NUMERIC(18, 4) DEFAULT 0,
  document_type TEXT,
  document_id TEXT,
  document_line_id TEXT,
  reference TEXT,
  legacy_transaction_id TEXT,
  created_by TEXT,
  notes TEXT,
  idempotency_key TEXT NOT NULL,
  CONSTRAINT stock_ledger_idempotency_unique UNIQUE (idempotency_key)
);

CREATE INDEX IF NOT EXISTS idx_stock_ledger_wh_product
  ON public.stock_ledger (warehouse_id, product_id);
CREATE INDEX IF NOT EXISTS idx_stock_ledger_occurred
  ON public.stock_ledger (occurred_at DESC);
CREATE INDEX IF NOT EXISTS idx_stock_ledger_legacy
  ON public.stock_ledger (legacy_transaction_id)
  WHERE legacy_transaction_id IS NOT NULL;

-- ---------------------------------------------------------------------------
-- On-hand view (read model)
-- ---------------------------------------------------------------------------
CREATE OR REPLACE VIEW public.v_stock_on_hand AS
SELECT
  warehouse_id,
  product_id,
  SUM(qty_delta)::NUMERIC(18, 4) AS quantity_on_hand,
  SUM(qty_delta * COALESCE(unit_cost, 0))::NUMERIC(18, 4) AS valuation_amount
FROM public.stock_ledger
GROUP BY warehouse_id, product_id
HAVING SUM(qty_delta) <> 0;

-- Global on-hand (all warehouses) for product-level screens
CREATE OR REPLACE VIEW public.v_stock_on_hand_product AS
SELECT
  product_id,
  SUM(qty_delta)::NUMERIC(18, 4) AS quantity_on_hand,
  SUM(qty_delta * COALESCE(unit_cost, 0))::NUMERIC(18, 4) AS valuation_amount
FROM public.stock_ledger
GROUP BY product_id
HAVING SUM(qty_delta) <> 0;

-- ---------------------------------------------------------------------------
-- RPC: post_stock_movement (single ledger insert, idempotent)
-- ---------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.post_stock_movement(
  p_warehouse_id TEXT,
  p_product_id TEXT,
  p_movement_type TEXT,
  p_qty_delta NUMERIC,
  p_unit_cost NUMERIC DEFAULT 0,
  p_document_type TEXT DEFAULT NULL,
  p_document_id TEXT DEFAULT NULL,
  p_document_line_id TEXT DEFAULT NULL,
  p_reference TEXT DEFAULT NULL,
  p_legacy_transaction_id TEXT DEFAULT NULL,
  p_created_by TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT now(),
  p_idempotency_key TEXT DEFAULT NULL
)
RETURNS UUID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_id UUID;
  v_key TEXT;
BEGIN
  IF p_warehouse_id IS NULL OR p_warehouse_id = '' THEN
    RAISE EXCEPTION 'warehouse_id is required';
  END IF;
  IF p_product_id IS NULL OR p_product_id = '' THEN
    RAISE EXCEPTION 'product_id is required';
  END IF;
  IF p_qty_delta IS NULL OR p_qty_delta = 0 THEN
    RAISE EXCEPTION 'qty_delta must be non-zero';
  END IF;

  v_key := COALESCE(
    p_idempotency_key,
    encode(sha256(
      (p_warehouse_id || '|' || p_product_id || '|' || p_movement_type || '|' ||
       p_qty_delta::TEXT || '|' || COALESCE(p_reference, '') || '|' ||
       COALESCE(p_legacy_transaction_id, '') || '|' ||
       COALESCE(p_document_line_id, ''))::bytea
    ), 'hex')
  );

  INSERT INTO public.stock_ledger (
    occurred_at, warehouse_id, product_id, movement_type, qty_delta,
    unit_cost, document_type, document_id, document_line_id, reference,
    legacy_transaction_id, created_by, notes, idempotency_key
  ) VALUES (
    COALESCE(p_occurred_at, now()), p_warehouse_id, p_product_id, p_movement_type,
    p_qty_delta, COALESCE(p_unit_cost, 0), p_document_type, p_document_id,
    p_document_line_id, p_reference, p_legacy_transaction_id, p_created_by,
    p_notes, v_key
  )
  ON CONFLICT (idempotency_key) DO NOTHING
  RETURNING id INTO v_id;

  IF v_id IS NULL THEN
    SELECT id INTO v_id FROM public.stock_ledger WHERE idempotency_key = v_key;
  END IF;

  RETURN v_id;
END;
$$;

-- ---------------------------------------------------------------------------
-- RPC: post_warehouse_movement (document-level: IN/OUT/TRANSFER/OIL)
-- ---------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.post_warehouse_movement(
  p_movement_kind TEXT,
  p_product_id TEXT,
  p_quantity NUMERIC,
  p_warehouse_id TEXT DEFAULT NULL,
  p_from_warehouse_id TEXT DEFAULT NULL,
  p_to_warehouse_id TEXT DEFAULT NULL,
  p_unit_cost NUMERIC DEFAULT 0,
  p_reference TEXT DEFAULT NULL,
  p_legacy_transaction_id TEXT DEFAULT NULL,
  p_created_by TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT now()
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_ids UUID[] := ARRAY[]::UUID[];
  v_id UUID;
  v_qty NUMERIC;
BEGIN
  v_qty := ABS(p_quantity);
  IF v_qty = 0 THEN
    RAISE EXCEPTION 'quantity must be positive';
  END IF;

  IF p_movement_kind = 'IN' THEN
    v_id := public.post_stock_movement(
      p_warehouse_id, p_product_id, 'IN', v_qty, p_unit_cost,
      'GRN', NULL, NULL, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'in:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

  ELSIF p_movement_kind IN ('OUT', 'OIL_OUT') THEN
    v_id := public.post_stock_movement(
      p_warehouse_id, p_product_id,
      CASE WHEN p_movement_kind = 'OIL_OUT' THEN 'OIL_OUT' ELSE 'OUT' END,
      -v_qty, p_unit_cost, 'GI', NULL, NULL, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'out:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

  ELSIF p_movement_kind = 'TRANSFER' THEN
    IF p_from_warehouse_id IS NULL OR p_to_warehouse_id IS NULL THEN
      RAISE EXCEPTION 'transfer requires from and to warehouse';
    END IF;
    IF p_from_warehouse_id = p_to_warehouse_id THEN
      RAISE EXCEPTION 'from and to warehouse must differ';
    END IF;

    v_id := public.post_stock_movement(
      p_from_warehouse_id, p_product_id, 'TRANSFER_OUT', -v_qty, p_unit_cost,
      'TRF', NULL, NULL, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'trf-out:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

    v_id := public.post_stock_movement(
      p_to_warehouse_id, p_product_id, 'TRANSFER_IN', v_qty, p_unit_cost,
      'TRF', NULL, NULL, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'trf-in:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);
  ELSE
    RAISE EXCEPTION 'unknown movement kind: %', p_movement_kind;
  END IF;

  RETURN jsonb_build_object('ledger_entry_ids', to_jsonb(v_ids));
END;
$$;

ALTER PUBLICATION supabase_realtime SET (publish = 'insert, update, delete');

-- ---------------------------------------------------------------------------
-- Kiá»ƒm tra publication
-- ---------------------------------------------------------------------------
SELECT 
  p.pubname AS publication_name,
  COUNT(t.schemaname || '.' || t.tablename) AS table_count,
  string_agg(t.schemaname || '.' || t.tablename, ', ') AS tables
FROM pg_publication p
LEFT JOIN pg_publication_tables t ON p.pubname = t.pubname
WHERE p.pubname = 'supabase_realtime'
GROUP BY p.pubname;


===== 006_migrate_to_employees_as_sso.sql =====

-- 006_migrate_to_employees_as_sso.sql
-- Migrate all user data and user_records data into employees as the single source of truth

-- ---------------------------------------------------------------------------
-- Step 1: Add missing columns to employees table
-- ---------------------------------------------------------------------------
ALTER TABLE public.employees
  ADD COLUMN IF NOT EXISTS role TEXT NOT NULL DEFAULT 'user',
  ADD COLUMN IF NOT EXISTS role_group TEXT,
  ADD COLUMN IF NOT EXISTS password_hash TEXT,
  ADD COLUMN IF NOT EXISTS last_login TIMESTAMPTZ;

-- ---------------------------------------------------------------------------
-- Step 2: Migrate data from users and user_records to employees
-- ---------------------------------------------------------------------------
-- Update existing employees with data from users/user_records
UPDATE public.employees e
SET
  role = COALESCE(u.role, e.role, 'user'),
  role_group = COALESCE(u.role_group, e.role_group),
  password_hash = COALESCE(ur.password_hash, e.password_hash),
  last_login = COALESCE(ur.last_login, e.last_login),
  updated_at = now()
FROM public.users u
LEFT JOIN public.user_records ur ON u.msnv = ur.msnv
WHERE e.msnv = u.msnv;

-- Insert employees from users/user_records that don't exist yet
INSERT INTO public.employees (
  msnv,
  ho_ten,
  phong_ban,
  chuc_vu,
  role,
  role_group,
  status,
  password_hash,
  last_login,
  created_at,
  updated_at
)
SELECT
  u.msnv,
  COALESCE(u.full_name, ur.full_name, u.msnv) AS ho_ten,
  COALESCE(u.department, ur.department) AS phong_ban,
  COALESCE(u.position, ur.position) AS chuc_vu,
  COALESCE(u.role, ur.role, 'user') AS role,
  u.role_group,
  COALESCE(u.status, CASE WHEN ur.status THEN 'active' ELSE 'inactive' END, 'active') AS status,
  ur.password_hash,
  ur.last_login,
  COALESCE(u.created_at, ur.created_at, now()) AS created_at,
  now() AS updated_at
FROM public.users u
LEFT JOIN public.user_records ur ON u.msnv = ur.msnv
LEFT JOIN public.employees e ON u.msnv = e.msnv
WHERE e.msnv IS NULL;

-- ---------------------------------------------------------------------------
-- Step 3: Ensure password_hash is NOT NULL for all employees
-- ---------------------------------------------------------------------------
UPDATE public.employees
SET password_hash = msnv
WHERE password_hash IS NULL;

ALTER TABLE public.employees
  ALTER COLUMN password_hash SET NOT NULL;

-- ---------------------------------------------------------------------------
-- Step 4: Update foreign key on user_permissions to reference employees.msnv
-- ---------------------------------------------------------------------------
ALTER TABLE public.user_permissions
  DROP CONSTRAINT IF EXISTS user_permissions_msnv_fkey;

ALTER TABLE public.user_permissions
  ADD CONSTRAINT user_permissions_msnv_fkey
  FOREIGN KEY (msnv) REFERENCES public.employees(msnv) ON DELETE CASCADE;

-- ----------------------------------------------------------------------------
-- 2) ThĂªm p_document_id + p_document_line_id + p_allow_negative vĂ o post_stock_movement
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.post_stock_movement(
  p_warehouse_id TEXT,
  p_product_id TEXT,
  p_movement_type TEXT,
  p_qty_delta NUMERIC,
  p_unit_cost NUMERIC DEFAULT 0,
  p_document_type TEXT DEFAULT NULL,
  p_document_id TEXT DEFAULT NULL,
  p_document_line_id TEXT DEFAULT NULL,
  p_reference TEXT DEFAULT NULL,
  p_legacy_transaction_id TEXT DEFAULT NULL,
  p_created_by TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT now(),
  p_idempotency_key TEXT DEFAULT NULL,
  p_allow_negative BOOLEAN DEFAULT FALSE
)
RETURNS UUID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_id UUID;
  v_key TEXT;
  v_current NUMERIC;
BEGIN
  IF p_warehouse_id IS NULL OR p_warehouse_id = '' THEN
    RAISE EXCEPTION 'warehouse_id is required';
  END IF;
  IF p_product_id IS NULL OR p_product_id = '' THEN
    RAISE EXCEPTION 'product_id is required';
  END IF;
  IF p_qty_delta IS NULL OR p_qty_delta = 0 THEN
    RAISE EXCEPTION 'qty_delta must be non-zero';
  END IF;

  v_key := COALESCE(
    p_idempotency_key,
    encode(sha256(
      (p_warehouse_id || '|' || p_product_id || '|' || p_movement_type || '|' ||
       p_qty_delta::TEXT || '|' || COALESCE(p_reference, '') || '|' ||
       COALESCE(p_legacy_transaction_id, '') || '|' ||
       COALESCE(p_document_line_id, ''))::bytea
    ), 'hex')
  );

  -- Idempotency check TRÆ¯á»C validate Ă¢m kho (retry khĂ´ng bá»‹ cháº·n)
  SELECT id INTO v_id FROM public.stock_ledger WHERE idempotency_key = v_key;
  IF v_id IS NOT NULL THEN RETURN v_id; END IF;

  -- Check Ă¢m kho
  IF p_qty_delta < 0 AND NOT p_allow_negative THEN
    SELECT COALESCE(SUM(qty_delta), 0) INTO v_current
      FROM public.stock_ledger
     WHERE warehouse_id = p_warehouse_id AND product_id = p_product_id;
    IF (v_current + p_qty_delta) < 0 THEN
      RAISE EXCEPTION
        'insufficient stock: warehouse=% product=% current=% delta=%',
        p_warehouse_id, p_product_id, v_current, p_qty_delta
        USING ERRCODE = 'check_violation';
    END IF;
  END IF;

  INSERT INTO public.stock_ledger (
    occurred_at, warehouse_id, product_id, movement_type, qty_delta,
    unit_cost, document_type, document_id, document_line_id, reference,
    legacy_transaction_id, created_by, notes, idempotency_key
  ) VALUES (
    COALESCE(p_occurred_at, now()), p_warehouse_id, p_product_id, p_movement_type,
    p_qty_delta, COALESCE(p_unit_cost, 0), p_document_type, p_document_id,
    p_document_line_id, p_reference, p_legacy_transaction_id, p_created_by,
    p_notes, v_key
  )
  ON CONFLICT (idempotency_key) DO NOTHING
  RETURNING id INTO v_id;

  IF v_id IS NULL THEN
    SELECT id INTO v_id FROM public.stock_ledger WHERE idempotency_key = v_key;
  END IF;

  RETURN v_id;
END;
$$;

-- ----------------------------------------------------------------------------
-- 3) post_warehouse_movement nháº­n thĂªm p_document_id (forward xuá»‘ng post_stock_movement)
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.post_warehouse_movement(
  p_movement_kind TEXT,
  p_product_id TEXT,
  p_quantity NUMERIC,
  p_warehouse_id TEXT DEFAULT NULL,
  p_from_warehouse_id TEXT DEFAULT NULL,
  p_to_warehouse_id TEXT DEFAULT NULL,
  p_unit_cost NUMERIC DEFAULT 0,
  p_reference TEXT DEFAULT NULL,
  p_legacy_transaction_id TEXT DEFAULT NULL,
  p_created_by TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT now(),
  p_document_id TEXT DEFAULT NULL,
  p_document_line_id TEXT DEFAULT NULL
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_ids UUID[] := ARRAY[]::UUID[];
  v_id UUID;
  v_qty NUMERIC;
BEGIN
  v_qty := ABS(p_quantity);
  IF v_qty = 0 THEN
    RAISE EXCEPTION 'quantity must be positive';
  END IF;

  IF p_movement_kind = 'IN' THEN
    v_id := public.post_stock_movement(
      p_warehouse_id, p_product_id, 'IN', v_qty, p_unit_cost,
      'GRN', p_document_id, p_document_line_id, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'in:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

  ELSIF p_movement_kind IN ('OUT', 'OIL_OUT') THEN
    v_id := public.post_stock_movement(
      p_warehouse_id, p_product_id,
      CASE WHEN p_movement_kind = 'OIL_OUT' THEN 'OIL_OUT' ELSE 'OUT' END,
      -v_qty, p_unit_cost, 'GI', p_document_id, p_document_line_id, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'out:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

  ELSIF p_movement_kind = 'TRANSFER' THEN
    IF p_from_warehouse_id IS NULL OR p_to_warehouse_id IS NULL THEN
      RAISE EXCEPTION 'transfer requires from and to warehouse';
    END IF;
    IF p_from_warehouse_id = p_to_warehouse_id THEN
      RAISE EXCEPTION 'from and to warehouse must differ';
    END IF;

    v_id := public.post_stock_movement(
      p_from_warehouse_id, p_product_id, 'TRANSFER_OUT', -v_qty, p_unit_cost,
      'TRF', p_document_id, p_document_line_id, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'trf-out:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);

    v_id := public.post_stock_movement(
      p_to_warehouse_id, p_product_id, 'TRANSFER_IN', v_qty, p_unit_cost,
      'TRF', p_document_id, p_document_line_id, p_reference, p_legacy_transaction_id,
      p_created_by, p_notes, p_occurred_at,
      'trf-in:' || COALESCE(p_legacy_transaction_id, p_reference) || ':' || p_product_id
    );
    v_ids := array_append(v_ids, v_id);
  ELSE
    RAISE EXCEPTION 'unknown movement kind: %', p_movement_kind;
  END IF;

  RETURN jsonb_build_object('ledger_entry_ids', to_jsonb(v_ids));
END;
$$;

  p_items        JSONB DEFAULT '[]'::jsonb
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_return_id UUID;
  v_item      JSONB;
  v_count     INT := 0;
BEGIN
  IF p_warehouse_id IS NULL OR p_warehouse_id = '' THEN
    RAISE EXCEPTION 'warehouse_id is required';
  END IF;

  IF p_items IS NULL OR jsonb_array_length(p_items) = 0 THEN
    RAISE EXCEPTION 'at least 1 item required';
  END IF;

  INSERT INTO public.tool_returns (
    warehouse_id, returned_by, received_by, returned_at,
    note, document_id, created_by
  ) VALUES (
    p_warehouse_id, p_returned_by, p_received_by,
    COALESCE(p_returned_at, now()),
    p_note, p_document_id, p_created_by
  ) RETURNING id INTO v_return_id;

  FOR v_item IN SELECT * FROM jsonb_array_elements(p_items)
  LOOP
    INSERT INTO public.tool_return_items (
      tool_return_id, product_id, qty_issued, qty_returned,
      condition, machine_id, project_id, note
    ) VALUES (
      v_return_id,
      v_item->>'product_id',
      COALESCE((v_item->>'qty_issued')::NUMERIC, 0),
      COALESCE((v_item->>'qty_returned')::NUMERIC, 0),
      NULLIF(v_item->>'condition', ''),
      NULLIF(v_item->>'machine_id', '')::UUID,
      NULLIF(v_item->>'project_id', '')::UUID,
      v_item->>'note'
    );
    v_count := v_count + 1;
  END LOOP;

  RETURN jsonb_build_object(
    'ok', true,
    'tool_return_id', v_return_id,
    'items_inserted', v_count
  );
END;
$$;

CREATE OR REPLACE FUNCTION public.audit_trigger_fn()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_user_id TEXT;
  v_old JSONB;
  v_new JSONB;
  v_record_id TEXT;
  v_changed TEXT[];
BEGIN
  IF TG_OP = 'DELETE' THEN
    v_old := to_jsonb(OLD);
    v_record_id := COALESCE(v_old ->> 'id', v_old ->> 'document_id', v_old ->> 'session_id', '');
  ELSIF TG_OP = 'UPDATE' THEN
    v_old := to_jsonb(OLD);
    v_new := to_jsonb(NEW);
    v_record_id := COALESCE(v_new ->> 'id', v_new ->> 'document_id', v_new ->> 'session_id', '');

    SELECT array_agg(keys.key ORDER BY keys.key)
      INTO v_changed
      FROM jsonb_object_keys(v_new) AS keys(key)
     WHERE (v_old -> keys.key) IS DISTINCT FROM (v_new -> keys.key);
  ELSE
    v_new := to_jsonb(NEW);
    v_record_id := COALESCE(v_new ->> 'id', v_new ->> 'document_id', v_new ->> 'session_id', '');
  END IF;

  v_user_id := COALESCE(
    NULLIF(current_setting('app.user_id', true), ''),
    CASE WHEN v_new IS NOT NULL THEN v_new ->> 'created_by' END,
    CASE WHEN v_old IS NOT NULL THEN v_old ->> 'created_by' END,
    'system'
  );

  INSERT INTO public.audit_logs (
    user_id, action, table_name, record_id, old_data, new_data, changed_fields
  )
  VALUES (
    v_user_id, TG_OP, TG_TABLE_NAME, v_record_id, v_old, v_new, v_changed
  );

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;
  RETURN NEW;
END;
$$;

DO $$
DECLARE
  table_name TEXT;
BEGIN
  FOREACH table_name IN ARRAY ARRAY[
    'stock_documents',
    'stock_document_lines',
    'stock_ledger',
    'tool_returns',
    'tool_return_items',
    'inventory_count_sessions',
    'inventory_count_lines'
  ]
  LOOP
    EXECUTE format('DROP TRIGGER IF EXISTS audit_%I ON public.%I', table_name, table_name);
    EXECUTE format(
      'CREATE TRIGGER audit_%I AFTER INSERT OR UPDATE OR DELETE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.audit_trigger_fn()',
      table_name,
      table_name
    );
  END LOOP;
END;
$$;

BEGIN;

-- Serialize movements for the same warehouse/product before choosing issue cost.
CREATE OR REPLACE FUNCTION public.post_warehouse_movement_weighted_average(
  p_movement_kind TEXT,
  p_product_id TEXT,
  p_quantity NUMERIC,
  p_warehouse_id TEXT DEFAULT NULL,
  p_from_warehouse_id TEXT DEFAULT NULL,
  p_to_warehouse_id TEXT DEFAULT NULL,
  p_unit_cost NUMERIC DEFAULT 0,
  p_reference TEXT DEFAULT NULL,
  p_legacy_transaction_id TEXT DEFAULT NULL,
  p_created_by TEXT DEFAULT NULL,
  p_notes TEXT DEFAULT NULL,
  p_occurred_at TIMESTAMPTZ DEFAULT now(),
  p_document_id TEXT DEFAULT NULL,
  p_document_line_id TEXT DEFAULT NULL
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_unit_cost NUMERIC := COALESCE(p_unit_cost, 0);
  v_on_hand NUMERIC;
BEGIN
  IF p_movement_kind IN ('OUT', 'OIL_OUT', 'TRANSFER') THEN
    PERFORM pg_advisory_xact_lock(hashtextextended(lock_keys.lock_key, 0))
      FROM (
        SELECT DISTINCT warehouse_id || '|' || p_product_id AS lock_key
          FROM unnest(
            CASE
              WHEN p_movement_kind = 'TRANSFER'
                THEN ARRAY[p_from_warehouse_id, p_to_warehouse_id]
              ELSE ARRAY[p_warehouse_id]
            END
          ) AS warehouses(warehouse_id)
         WHERE warehouse_id IS NOT NULL
         ORDER BY lock_key
      ) AS lock_keys;

    SELECT SUM(qty_delta),
           CASE
             WHEN SUM(qty_delta) > 0
               THEN SUM(qty_delta * COALESCE(unit_cost, 0)) / SUM(qty_delta)
             ELSE NULL
           END
      INTO v_on_hand, v_unit_cost
      FROM public.stock_ledger
     WHERE warehouse_id = CASE
         WHEN p_movement_kind = 'TRANSFER' THEN p_from_warehouse_id
         ELSE p_warehouse_id
       END
       AND product_id = p_product_id;

    IF COALESCE(v_on_hand, 0) <= 0 OR v_unit_cost IS NULL THEN
      RAISE EXCEPTION
        'Cannot determine weighted-average cost: warehouse=% product=% on_hand=%',
        CASE WHEN p_movement_kind = 'TRANSFER' THEN p_from_warehouse_id ELSE p_warehouse_id END,
        p_product_id,
        COALESCE(v_on_hand, 0);
    END IF;
  END IF;

  RETURN public.post_warehouse_movement(
    p_movement_kind => p_movement_kind,
    p_product_id => p_product_id,
    p_quantity => p_quantity,
    p_warehouse_id => p_warehouse_id,
    p_from_warehouse_id => p_from_warehouse_id,
    p_to_warehouse_id => p_to_warehouse_id,
    p_unit_cost => v_unit_cost,
    p_reference => p_reference,
    p_legacy_transaction_id => p_legacy_transaction_id,
    p_created_by => p_created_by,
    p_notes => p_notes,
    p_occurred_at => p_occurred_at,
    p_document_id => p_document_id,
    p_document_line_id => p_document_line_id
  );
END;
$$;

REVOKE ALL ON FUNCTION public.post_warehouse_movement_weighted_average(
  TEXT, TEXT, NUMERIC, TEXT, TEXT, TEXT, NUMERIC, TEXT, TEXT, TEXT, TEXT,
  TIMESTAMPTZ, TEXT, TEXT
) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.post_warehouse_movement_weighted_average(

CREATE OR REPLACE FUNCTION public.get_valuation_history(
  p_month_from TEXT,
  p_month_to TEXT
)
RETURNS TABLE (
  month TEXT,
  warehouse_id TEXT,
  product_id TEXT,
  closing_qty NUMERIC,
  closing_value NUMERIC
)
LANGUAGE plpgsql
STABLE
SET search_path = pg_catalog, public
AS $$
DECLARE
  v_month_from DATE;
  v_month_to DATE;
BEGIN
  IF p_month_from !~ '^[0-9]{4}-(0[1-9]|1[0-2])$'
    OR p_month_to !~ '^[0-9]{4}-(0[1-9]|1[0-2])$' THEN
    RAISE EXCEPTION 'Month values must use YYYY-MM format';
  END IF;

  v_month_from := (p_month_from || '-01')::DATE;
  v_month_to := (p_month_to || '-01')::DATE;

  IF v_month_from > v_month_to THEN
    RAISE EXCEPTION 'Start month must not be after end month';
  END IF;
  IF v_month_to >= v_month_from + INTERVAL '120 months' THEN
    RAISE EXCEPTION 'Valuation history range cannot exceed 120 months';
  END IF;

  RETURN QUERY
  WITH months AS (
    SELECT series.month_start::DATE AS month_start
      FROM generate_series(
        v_month_from::TIMESTAMP,
        v_month_to::TIMESTAMP,
        INTERVAL '1 month'
      ) AS series(month_start)
  ),
  dimensions AS (
    SELECT DISTINCT l.warehouse_id, l.product_id
      FROM public.stock_ledger AS l
     WHERE l.occurred_at < (v_month_to + INTERVAL '1 month')
  ),
  opening AS (
    SELECT l.warehouse_id,
           l.product_id,
           SUM(l.qty_delta) AS qty,
           SUM(l.qty_delta * COALESCE(l.unit_cost, 0)) AS value
      FROM public.stock_ledger AS l
     WHERE l.occurred_at < v_month_from::TIMESTAMPTZ
     GROUP BY l.warehouse_id, l.product_id
  ),
  monthly_movements AS (
    SELECT l.warehouse_id,
           l.product_id,
           date_trunc('month', l.occurred_at)::DATE AS month_start,
           SUM(l.qty_delta) AS qty,
           SUM(l.qty_delta * COALESCE(l.unit_cost, 0)) AS value
      FROM public.stock_ledger AS l
     WHERE l.occurred_at >= v_month_from::TIMESTAMPTZ
       AND l.occurred_at < (v_month_to + INTERVAL '1 month')::TIMESTAMPTZ
     GROUP BY l.warehouse_id, l.product_id, date_trunc('month', l.occurred_at)::DATE
  ),
  period_grid AS (
    SELECT m.month_start,
           d.warehouse_id,
           d.product_id,
           COALESCE(o.qty, 0) AS opening_qty,
           COALESCE(o.value, 0) AS opening_value,
           COALESCE(mm.qty, 0) AS movement_qty,
           COALESCE(mm.value, 0) AS movement_value
      FROM months AS m
      CROSS JOIN dimensions AS d
      LEFT JOIN opening AS o
        ON o.warehouse_id = d.warehouse_id AND o.product_id = d.product_id
      LEFT JOIN monthly_movements AS mm
        ON mm.month_start = m.month_start
       AND mm.warehouse_id = d.warehouse_id
       AND mm.product_id = d.product_id
  ),
  closing AS (
    SELECT month_start,
           warehouse_id,
           product_id,
           opening_qty + SUM(movement_qty) OVER (
             PARTITION BY warehouse_id, product_id ORDER BY month_start
           ) AS closing_qty,
           opening_value + SUM(movement_value) OVER (
             PARTITION BY warehouse_id, product_id ORDER BY month_start
           ) AS closing_value
      FROM period_grid
  )
  SELECT to_char(c.month_start, 'YYYY-MM'),
         c.warehouse_id,
         c.product_id,
         c.closing_qty,
         c.closing_value
    FROM closing AS c
   ORDER BY c.month_start, c.warehouse_id, c.product_id;
END;
$$;

GRANT EXECUTE ON FUNCTION public.get_valuation_history(TEXT, TEXT)
  TO anon, authenticated;

COMMIT;
