export const isNavItemVisible = (
  item: ERPNavItem | undefined,
  user?: ERPNavAuthUser
) => {
  if (!user || !item) return false;
  
  // Admin full access
  if (user.role === 'admin') return true;
  
  // KhĂ´ng cĂ³ permission key = má»i ngÆ°á»i Ä‘á»u tháº¥y
  if (!item.permissionKey) return true;

  // Kiá»ƒm tra trá»±c tiáº¿p tá»« user.permissions[permissionKey]
  const hasPermission = hasPermissionFlag(user.permissions?.[item.permissionKey], 'view');
  
  return hasPermission;
};
